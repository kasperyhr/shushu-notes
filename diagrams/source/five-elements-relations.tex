% 五行生克关系图
% 用途：展示五行（木火土金水）的相生和相克关系
\documentclass[tikz,border=10pt]{standalone}
\usepackage{ctex}
\usepackage{xcolor}
\pagecolor{white}
\usetikzlibrary{arrows.meta, positioning, calc, decorations.pathmorphing, backgrounds}

\begin{document}
\begin{tikzpicture}[scale=1.4]
  % 五行节点坐标（五边形排列）
  \coordinate (mu)   at ( 90:3.2);
  \coordinate (huo)  at ( 18:3.2);
  \coordinate (tu)   at (306:3.2);
  \coordinate (jin)  at (234:3.2);
  \coordinate (shui) at (162:3.2);

  % 相生：外圈绿色弧形箭头（加粗）
  \draw[green!50!black, ultra thick, ->, >=Stealth] (mu)   to[bend left=50] node[font=\normalsize\bfseries, midway, above right, fill=white, inner sep=2pt] {生} (huo);
  \draw[green!50!black, ultra thick, ->, >=Stealth] (huo)  to[bend left=30] node[font=\normalsize\bfseries, midway, right, fill=white, inner sep=2pt] {生} (tu);
  \draw[green!50!black, ultra thick, ->, >=Stealth] (tu)   to[bend left=30] node[font=\normalsize\bfseries, midway, below left, fill=white, inner sep=2pt] {生} (jin);
  \draw[green!50!black, ultra thick, ->, >=Stealth] (jin)  to[bend left=30] node[font=\normalsize\bfseries, midway, left, fill=white, inner sep=2pt] {生} (shui);
  \draw[green!50!black, ultra thick, ->, >=Stealth] (shui) to[bend left=30] node[font=\normalsize\bfseries, midway, above left, fill=white, inner sep=2pt] {生} (mu);

  % 相克：内部星形红色箭头（加粗）
  \draw[red!70!black, ultra thick, ->, >=Stealth] (mu)   -- node[font=\normalsize\bfseries, midway, above right, fill=white, inner sep=2pt] {克} (tu);
  \draw[red!70!black, ultra thick, ->, >=Stealth] (tu)   -- node[font=\normalsize\bfseries, midway, below, fill=white, inner sep=2pt] {克} (shui);
  \draw[red!70!black, ultra thick, ->, >=Stealth] (shui) -- node[font=\normalsize\bfseries, midway, left, fill=white, inner sep=2pt] {克} (huo);
  \draw[red!70!black, ultra thick, ->, >=Stealth] (huo)  -- node[font=\normalsize\bfseries, midway, right, fill=white, inner sep=2pt] {克} (jin);
  \draw[red!70!black, ultra thick, ->, >=Stealth] (jin)  -- node[font=\normalsize\bfseries, midway, above left, fill=white, inner sep=2pt] {克} (mu);

  % 五行节点（加大）
  \fill[green!60!black] (mu) circle (14pt);
  \node[font=\Large\bfseries, white] at (mu) {木};

  \fill[red!60!black] (huo) circle (14pt);
  \node[font=\Large\bfseries, white] at (huo) {火};

  \fill[yellow!60!black] (tu) circle (14pt);
  \node[font=\Large\bfseries, black] at (tu) {土};

  \fill[gray!60!black] (jin) circle (14pt);
  \node[font=\Large\bfseries, white] at (jin) {金};

  \fill[blue!60!black] (shui) circle (14pt);
  \node[font=\Large\bfseries, white] at (shui) {水};

  % 图例（放大）
  \node[font=\normalsize, green!50!black, anchor=west] at (-4.8,-4.2) {外圈 \textbf{相生}：提供条件、滋养、增强};
  \node[font=\normalsize, red!70!black, anchor=west]  at (-4.8,-4.8) {内部 \textbf{相克}：约束、控制、制衡};

  % 白色背景，防止 GitHub dark mode 下透明 SVG 看不清
  \begin{scope}[on background layer]
    \fill[white] (current bounding box.south west) rectangle (current bounding box.north east);
  \end{scope}
\end{tikzpicture}
\end{document}
